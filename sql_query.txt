7: Write a sql query to find employees who has no managerial position;

FROM emplo AS e
LEFT JOIN emplo AS p
ON e.employee_id = p.manager_id
WHERE p.employee_id IS NULL;

18: For order ID 1, list the names of the products ordered and their quantities.

SELECT
    p.name AS product_name,
    oi.quantity
FROM order_items oi
JOIN products p
ON oi.product_id = p.id
WHERE oi.order_id = 1;


19: given user id name created all user created in last 7 + count registered on createion date


SELECT
    id,
    name,
    created_at,
    COUNT(*) OVER (PARTITION BY CAST(created_at AS DATE)) AS registered_on_creation_date
FROM Users
WHERE created_at >= DATEADD(DAY, -7, CAST(GETDATE() AS DATE))
ORDER BY created_at DESC;
